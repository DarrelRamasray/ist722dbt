REM Paste ONE line at a time and press Enter after each.
REM What it does: uploads employeesalary.csv to the Milestone 2 stage

REM Before this: employeesalary.csv must be in
REM   C:\Users\darre\Documents\chinook_extract (guide Part 1).

chcp 65001

cd %USERPROFILE%\Documents\chinook_extract

snow sql -c my_snowflake -q "PUT 'file://C:/Users/darre/Documents/chinook_extract/employeesalary.csv' @RAW.CHINOOK.CHINOOK_STAGE AUTO_COMPRESS=TRUE OVERWRITE=TRUE"
