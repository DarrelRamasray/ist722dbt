-- ---------------------------------------------------------------------
-- 1 of 2  ->  playlist.csv  (18 rows)
-- ---------------------------------------------------------------------
SELECT PlaylistId, Name
FROM Chinook.dbo.Playlist
ORDER BY PlaylistId;


-- ---------------------------------------------------------------------
-- 2 of 2  ->  playlisttrack.csv  (8,715 rows)
-- ---------------------------------------------------------------------
SELECT PlaylistId, TrackId
FROM Chinook.dbo.PlaylistTrack
ORDER BY PlaylistId, TrackId;
